#   with
#       from packaging.version import parse as parse_version
#   then `pip install --no-deps .` (its metadata pins numpy 1.23 / sklearn 1.2).
#   HeatGeo also needs the compiled s_gd2 extension, which is architecture-
#   specific (no Linux/py3.11 wheel; painful on Apple Silicon).
